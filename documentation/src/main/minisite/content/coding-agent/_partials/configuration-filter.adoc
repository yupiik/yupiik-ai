<label for="config-filter">Filter configuration options by key</label>
<input type="search" id="config-filter" placeholder="For example: provider, mcp or contextWindow" aria-describedby="config-filter-status" autocomplete="off">
<p id="config-filter-status" role="status" aria-live="polite"></p>
<script>
(function () {
  var input = document.getElementById('config-filter');
  var polish = function (s) { return (s || '').toLowerCase().replace(/[^a-z0-9.]+/g, ''); };
  var apply = function () {
    var q = polish(input.value);
    var totalVisible = 0;
    document.querySelectorAll('.config-list').forEach(function (list) {
      var visible = 0;
      list.querySelectorAll('dl').forEach(function (dl) {
        dl.querySelectorAll('dt.hdlist1').forEach(function (dt) {
          var dd = dt.nextElementSibling;
          var show = !q || polish(dt.textContent).indexOf(q) !== -1;
          dt.style.display = show ? '' : 'none';
          if (dd && dd.tagName === 'DD') { dd.style.display = show ? '' : 'none'; }
          if (show) { visible++; }
        });
      });
      var noMatch = list.querySelector('.config-no-match');
      if (visible === 0) {
        if (!noMatch) {
          noMatch = document.createElement('p');
          noMatch.className = 'config-no-match';
          noMatch.textContent = 'No option matches the filter.';
          list.appendChild(noMatch);
        }
        noMatch.style.display = '';
      } else if (noMatch) {
        noMatch.style.display = 'none';
      }
      totalVisible += visible;
    });
    document.getElementById('config-filter-status').textContent = totalVisible + ' matching options.';
  };
  input.addEventListener('input', apply);
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', apply, { once: true });
  } else {
    apply();
  }
})();
</script>
